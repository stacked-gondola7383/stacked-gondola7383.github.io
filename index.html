sign.ts
import { hmac } from "e7:crypto";
import { hexEncode } from "e7:encoding";
import type { KeyHandle } from "e7:crypto";

// The key lives in the Keychain / Keystore.
// The script can use it, but never see it.
export function sign(params: Map<string, string>,
                     key: KeyHandle): string {
  const enc = (s: string) => encodeURIComponent(s);
  const query = [...params.keys()].sort()
    .map(k => `${enc(k)}=${enc(params.get(k)!)}`)
    .join("&");
  const data = new TextEncoder().encode(query);
  return hexEncode(hmac("sha256", key, data));
}

Why E7

Update algorithms without shipping a new app

Request signing, webhook checks and license validation change faster than app releases. E7 moves that logic into small scripts you compile and sign on your server.

⚡

Ship without a release

Compile and sign on your build server, deliver .e7b bytecode over your CDN or API, and the app verifies and runs it.

🛡️

Safe by construction

Signature check, then a bytecode verifier that guarantees memory safety even for tampered input, then budgets for instructions, memory and call depth.

🔑

Secrets stay in the host

A KeyHandle lets a script use a key held in the Keychain, Keystore or DPAPI without ever reading the key bytes.

🧩

TypeScript you already know

Every E7 program is valid TypeScript. Use VS Code and tsc --strict; results match Node.js, checked by differential tests.

🔐

Batteries for crypto

SHA-2/3, HMAC, HKDF, PBKDF2, AES-GCM, ChaCha20-Poly1305, Ed25519, X25519, ECDSA and RSA, running as native code.

📦

Small and fast

About 470 KB for the runtime, 920 KB with the on-device compiler. The register VM matches or beats V8's interpreter.

How it works

From TypeScript to a verified, sandboxed call

Your server compiles and signs. Your app trusts only your public key, and nothing runs until the signature and the bytecode have both been checked.

Write

Write a script in the E7 subset of TypeScript, with full type checking in your editor.

Compile & sign

e7c type-checks, compiles to compact bytecode, and signs it with your Ed25519 key.

Verify & run

e7vm checks the signature, verifies the bytecode, links your host functions, then runs it under a budget.

Language tour

Strict types, JavaScript semantics

No any, no surprises: names and types are resolved at compile time, numbers behave exactly like JavaScript, and fixed-width integers are there when an algorithm needs them. Every output below is real.

Interfaces, classes, narrowing

Interfaces, classes with single inheritance, closures, destructuring, optional chaining and nullish coalescing, all statically checked.

  • T | null | undefined nullable types
  • Narrowing on === null, truthiness, instanceof
  • Multi-file import / export
ledger.ts
interface Order { id: string; amount: number; tags?: string[] }

class Ledger {
  private items: Order[] = [];
  add(o: Order): void { this.items.push(o); }
  total(): number { return this.items.reduce((s, o) => s + o.amount, 0); }
  find(id: string): Order | null { return this.items.find(o => o.id === id) ?? null; }
}

const ledger = new Ledger();
ledger.add({ id: "A1", amount: 42.5 });
ledger.add({ id: "B2", amount: 7.5, tags: ["promo"] });
const o = ledger.find("B2");
if (o !== null) console.log(o.id, o.tags?.length ?? 0);   // narrowed: o is Order here
console.log(ledger.total());
$ e7 ledger.ts B2 1 50

Machine integers, on request

number is a 64-bit float exactly as in JavaScript. When an algorithm needs real machine integers, use i8 … u64: they wrap on overflow and compile to single instructions.

  • u32(x), Number(x): explicit conversions only
  • Intrinsics: rotl32, clz32, bswap32, mulhi64
fnv.ts
// number is IEEE-754 like JS; fixed-width integers are opt-in and wrap like machine ints
function fnv1a(data: Uint8Array): u32 {
  let h: u32 = 0x811c9dc5;
  for (const b of data) {
    h ^= u32(b);
    h *= 16777619;          // wraps modulo 2^32
  }
  return h;
}
console.log(fnv1a(new TextEncoder().encode("hello")).toString(16));
console.log(rotl32(u32(1), 31), clz32(u32(1)));
$ e7 fnv.ts 4f9f2cab 2147483648 31

JSON that checks itself

JSON.parse(text) as T validates the data against T while parsing. A wrong shape stops the script with the exact path, instead of crashing later.

  • Nested objects, arrays, tuples, optional fields
  • Map<string, V> for dynamic keys
payment.ts
interface Payment { orderId: string; amount: number; items: { sku: string; qty: i32 }[] }

const p = JSON.parse('{"orderId":"A1","amount":12.5,"items":[{"sku":"X","qty":2}]}') as Payment;
console.log(p.items[0].qty * 2);   // validated against the type at parse time

const bad = JSON.parse('{"orderId":"A1","amount":"12.5","items":[]}') as Payment;
$ e7 payment.ts 4 JSON.parse: expected a number at $.amount at <init> (payment.ts:6)

Signing and verification, built in

e7:crypto and e7:encoding run natively (BearSSL and Monocypher). Scripts only assemble parameters, so a typical call takes well under a millisecond.

  • Hashes, HMAC, HKDF, PBKDF2, AES-GCM/CBC/CTR, ChaCha20-Poly1305
  • Ed25519, X25519, ECDSA P-256/384/521, RSA PKCS#1 v1.5 & PSS
  • hex, base64, base64url, base32, UTF-8, PEM
verify.ts
import { ed25519Verify, hmac, sha256 } from "e7:crypto";
import { base64UrlDecode, hexEncode } from "e7:encoding";

// Verify a webhook: HMAC-SHA256 over "timestamp.body"
export function verifyWebhook(secret: Uint8Array, ts: string, body: string, sigHex: string): boolean {
  const enc = new TextEncoder();
  return hexEncode(hmac("sha256", secret, enc.encode(`${ts}.${body}`))) === sigHex;
}

// Verify an Ed25519-signed license token: base64url(payload).base64url(signature)
export function checkLicense(publicKey: Uint8Array, token: string): boolean {
  const [payload, sig] = token.split(".");
  return ed25519Verify(publicKey, base64UrlDecode(payload), base64UrlDecode(sig));
}

console.log(hexEncode(sha256(new TextEncoder().encode("abc"))).slice(0, 16));
console.log(verifyWebhook(new TextEncoder().encode("k"), "1700000000", "{}", "00"));
$ e7 verify.ts ba7816bf8f01cfea false
Security

Built to run code that came over the network

Every layer assumes the bytecode may be hostile. E7 went through four independent security reviews, differential testing against Node.js and OpenSSL, sanitizers and fuzzing, and every finding was fixed with a regression test.

  • Ed25519 module signaturesOnly modules signed by your trusted public keys load; the check happens before parsing.
  • Bytecode verifierRegisters, kinds, indices, jump targets and call metadata are validated, so even tampered bytecode cannot corrupt memory.
  • Budgets and limitsAn instruction budget (heavy native calls included), a memory limit and a call-depth limit stop runaway scripts.
  • Explicit capabilitiesScripts can call only the host functions you register; there is no file, network or system access.
  • Keys stay in the hostKeyHandle resolves keys natively; a script can sign with a key it can never read or leak.
  • No JITA pure interpreter: no executable memory, which keeps it within App Store rules.

Evidence and every review finding: docs/PRODUCTION.md

Getting started

From zero to a verified call in five minutes

You need CMake 3.16+ and a C++17 compiler (GCC, Clang or MSVC). Every output shown is real, produced by examples/embed in the repository.

1Build

Produces e7 (run), e7c (compile, sign, disassemble) and the static library.

terminal
git clone https://github.com/e7lang/e7.git && cd e7
cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
cmake --build build -j

2Write a script

Exported functions are what your app calls. main runs when you use the CLI.

examples/embed/sign.ts
import { hmac } from "e7:crypto";
import { hexEncode } from "e7:encoding";

// Called by the host app: returns hex(HMAC-SHA256(secret, message))
export function sign(message: string, secret: string): string {
  const enc = new TextEncoder();
  return hexEncode(hmac("sha256", enc.encode(secret), enc.encode(message)));
}

export function main(): string {
  return sign("amount=100&currency=CNY", "s3cr3t");
}

3Run it

e7 compiles in memory and calls main (or --call fn).

terminal
cd examples/embed
../../build/e7 sign.ts
0fa1ee36e152de30350b6a463f4f365d6574bea59f99ccfebb599c7b6b1f1fd6

4Compile & sign

Keep release.seed on your build server; build release.pub into the app.

terminal
../../build/e7c --keygen release                  # release.seed (secret) + release.pub
../../build/e7c sign.ts -o sign.e7b --sign release.seed
../../build/e7 --trust release.pub sign.e7b       # runs only if signed by this key

5Call it from your app

Same flow on every platform: create a VM that trusts your key, load the bytecode, call an export.

host.c
#include "e7/e7.h"

e7_config cfg;
e7_config_init(&cfg);
cfg.require_signature = 1;            /* only run modules signed by our key */
cfg.trusted_keys = pub;               /* 32-byte Ed25519 public key(s) baked into the app */
cfg.num_trusted_keys = 1;
cfg.memory_limit = 32u << 20;
e7_vm* vm = e7_vm_new(&cfg);

e7_module* mod;
if (e7_load(vm, code, len, &mod) != E7_OK) { puts(e7_last_error(vm)); return 1; }

e7_value args[2];
args[0].type = E7_STRING; args[0].v.str.ptr = "amount=100&currency=CNY"; args[0].v.str.len = 23;
args[1].type = E7_STRING; args[1].v.str.ptr = "s3cr3t";                  args[1].v.str.len = 6;
e7_value ret;
e7_set_budget(vm, 10000000);          /* stop runaway scripts */
if (e7_call(vm, mod, "sign", args, 2, &ret) == E7_OK)
  printf("%.*s\n", (int)ret.v.str.len, ret.v.str.ptr);
e7_vm_free(vm);
Android · Kotlin
import org.e7vm.*

val vm = E7VM(E7VM.Config().requireSignature(SERVER_PUBLIC_KEY).memoryLimit(32L shl 20))
vm.setKeyProvider { id -> keyStoreDecrypt(id) }          // returns a ByteArray
vm.registerNative("deviceId", ">s") { _ -> Settings.Secure.getString(cr, ANDROID_ID) }
val module = vm.load(bytesFromServer)                      // throws E7Exception if verification fails
vm.setBudget(10_000_000)
val sig = vm.call(module, "sign", "a=1&b=2", KeyHandle(1)) as String
vm.close()
iOS · Swift
let vm = try E7VM(memoryLimit: 32 << 20, trustedKeys: [serverPublicKey])
vm.keyProvider = { id in Keychain.key(for: id) }
try vm.registerNative("deviceId", signature: ">s") { _ in .string(UIDevice.current.identifierForVendor!.uuidString) }
let module = try vm.load(bytecode)
vm.setBudget(10_000_000)
if case let .string(sig) = try vm.call(module, "sign", .string("a=1&b=2"), .handle(KeyHandle(1))) { print(sig) }

using var vm = new E7.E7VM(memoryLimit: 32 << 20, suspendOnBudget: false, serverPublicKey);
vm.SetKeyProvider(id => ProtectedData.Unprotect(LoadBlob(id), null, DataProtectionScope.CurrentUser));
var module = vm.Load(File.ReadAllBytes("app.e7b"));
var sig = (string)vm.Call(module, "sign", "a=1&b=2", new E7.KeyHandle(1));

Editor setup: add lib/e7.d.ts to your tsconfig.json for completion and type errors in VS Code.

Performance

Interpreter speed where it matters

On iOS every engine must interpret. Compared with V8 in the same mode (--jitless), e7vm is faster on most workloads, and standard crypto always runs as native code.
